# Official upstream build; local options avoid altering parent cache defaults.
set(BUILD_EXAMPLES OFF)
set(BUILD_PYTHON_MODULE OFF)
set(BUILD_CLOUD_CLIENT OFF)
set(BUILD_TESTS OFF)
set(BUILD_BENCHMARK OFF)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(vendor EXCLUDE_FROM_ALL)
target_include_directories(ruckig SYSTEM INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/vendor/include>")
add_library(relic_ruckig_qualification_core STATIC qualification.cpp)
target_link_libraries(relic_ruckig_qualification_core PUBLIC ruckig::ruckig)
target_compile_options(relic_ruckig_qualification_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(relic_ruckig_qualification qualification_main.cpp)
target_link_libraries(relic_ruckig_qualification PRIVATE relic_ruckig_qualification_core)
add_library(relic_ruckig_adapter STATIC adapter.cpp)
target_link_libraries(relic_ruckig_adapter PUBLIC relic_sim ruckig::ruckig)
target_compile_options(relic_ruckig_adapter PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
foreach(target IN ITEMS benchmark adapter_tests)
  add_executable(relic_ruckig_${target} ${target}.cpp)
  target_link_libraries(relic_ruckig_${target} PRIVATE relic_ruckig_adapter)
  target_compile_options(relic_ruckig_${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endforeach()
target_link_libraries(relic_ruckig_benchmark PRIVATE relic_ruckig_qualification_core)
if(BUILD_TESTING)
  add_test(NAME ruckig_qualification COMMAND relic_ruckig_qualification
    "${CMAKE_CURRENT_BINARY_DIR}/qualification.csv")
  set_tests_properties(ruckig_qualification PROPERTIES TIMEOUT 60 FIXTURES_SETUP ruckig_qualified)
  add_test(NAME ruckig_adapter_execution COMMAND relic_ruckig_adapter_tests "${PROJECT_SOURCE_DIR}/examples")
  set_tests_properties(ruckig_adapter_execution PROPERTIES TIMEOUT 120 FIXTURES_REQUIRED ruckig_qualified)
  add_test(NAME ruckig_benchmark COMMAND relic_ruckig_benchmark "${PROJECT_SOURCE_DIR}/examples"
    "${CMAKE_CURRENT_BINARY_DIR}/benchmark")
  set_tests_properties(ruckig_benchmark PROPERTIES TIMEOUT 180 FIXTURES_REQUIRED ruckig_qualified DEPENDS ruckig_adapter_execution)
endif()
