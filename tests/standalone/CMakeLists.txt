cmake_minimum_required(VERSION 3.24)
project(RELICPathSanitizerProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MUJOCO_ROOT "" CACHE PATH "Headers only: no vendor DLL linked into standalone tests")
set(RELIC_SANITIZER "" CACHE STRING "GCC sanitizer: address or undefined; empty for plain build")
option(RELIC_UBSAN_TRAP "Use UBSan trap mode without the external runtime" OFF)
set(RELIC_FEASIBILITY_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/feasibility.cpp" CACHE FILEPATH "Validator source; override only for isolated mutation copies")
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS "${MUJOCO_ROOT}/include" REQUIRED)
add_executable(relic_path_standalone ../path_campaign.cpp "${RELIC_FEASIBILITY_SOURCE}")
target_include_directories(relic_path_standalone PRIVATE ../../include "${MUJOCO_INCLUDE_DIR}")
target_compile_definitions(relic_path_standalone PRIVATE RELIC_STANDALONE)
target_compile_options(relic_path_standalone PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(RELIC_SANITIZER)
  target_compile_options(relic_path_standalone PRIVATE "-fsanitize=${RELIC_SANITIZER}" -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(relic_path_standalone PRIVATE "-fsanitize=${RELIC_SANITIZER}" -fno-sanitize-recover=all)
endif()
if(RELIC_UBSAN_TRAP)
  target_compile_options(relic_path_standalone PRIVATE -fsanitize=undefined -fsanitize-undefined-trap-on-error)
  target_link_options(relic_path_standalone PRIVATE -fsanitize=undefined -fsanitize-undefined-trap-on-error)
endif()
